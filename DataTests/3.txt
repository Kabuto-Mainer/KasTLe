use print
use scan

main {
    var i32 x = cast <i32> 10;

    scanf("%d", &x);
    printf("X = %d\n", x);
    x = fact(x);
    printf("Fact X = %d\n", x);
    printf("Poltorashka\n");
    exit;
}

func i32 fact(i32 a) {
    if (a == 1) {
        return cast <i32> 1;
    }
    return a * fact( a - cast <i32> 1);
}
