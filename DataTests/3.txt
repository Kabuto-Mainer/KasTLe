typedef i8 -> stack_value

var const i32 MAX_STACK_SIZE = 10;

block stack_t {
    i32     head;
    i32     size;
    stack_value[10]  data;
}

func bool stack_init(stack_t* stk,
                    i32 size)
{
    if (size <= 0 or size > MAX_STACK_SIZE) {
        return cast <bool> 0;
    }

    for (var i32 i = 0; i < MAX_STACK_SIZE; i = i + 1) {
        stk->data[i] = cast <stack_value> 0;
    }

    stk->head = 0;
    stk->size = size;

    return cast <bool> 0;
}
