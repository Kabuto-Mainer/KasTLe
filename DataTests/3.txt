use i32 printf(const char* format, i32 a1, i32 a2, i32 a3);
use i32* calloc(i64 n, i64 size);
use void free(i64* adr);



main {
    var i32 x = 10;
    x = fact(x);
    printf("FACT: %s\n", x, 0, 0);
    exit;
}

func i32 fact(i32 a) {
    if (a == 1) {
        return 1;
    }
    return fact(a - 1);
}
